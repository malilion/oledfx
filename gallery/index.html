import { createFramebuffer, effects } from './src/index.js';

const screen = createFramebuffer('ssd1306-128x64');
const fx = effects['eyes/happy-squint'].create({
  seed: 1, lookX: 0.4, lookY: -0.2
});
fx.step(1 / 16);
fx.draw(screen); // screen.data -> 1024-byte SSD1306 buffer
